/* v0.226 (task D3): the type scale C1. Four sizes, three weights, two families, all defined here and nowhere else.
   Every font size, weight and family in the CSS of the app refers to these tokens (tools/edits/taskD3.py put them in by rule, the mapping is in docs/backlog-log.md).
   --fs1 11 px: labels, values, small buttons. --fs2 12 px: body text of dense panels, names. --fs3 13 px: the app's base size, menus, text in windows. --fs4 15 px: titles and large readouts.
   --fw4 400 text, --fw5 500 medium, --fw6 600 names, values and titles. --ffu Inter for everything, --fft mono only where time is shown (bar position, time code, clocks).
   Digits have equal width everywhere (tabular), so numbers do not jitter while they change. Kept outside the scale: the logo wordmark, the splash, glyphs used as icons, text drawn inside pictures. */

:where(b,strong,th,h1,h2,h3,h4,h5,h6,optgroup){font-weight:var(--fw6)}
:where(small,sub,sup,h5,h6){font-size:var(--fs1)}
:where(h1,h2,h3,h4){font-size:var(--fs4)}
:where(kbd,code,samp,pre,textarea){font-family:var(--ffu)}
:where(textarea){font-size:var(--fs3)}
.t-time{font-family:var(--fft)}
.lcd.tempo,.lcd.tempo input,.mbpm,#tmw .tmsl .tmflt{font-family:var(--ffu)}
